# Automatic bpf_for Lowering — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the compiler plugin automatically lower dynamic-bound canonical `for` loops and dynamic-condition `while` loops to `bpf_for` / `while (can_loop)` so the BPF verifier accepts them without users touching `@BoundedBy` or `bpfLoop`.

**Architecture:** A new `LoopShapeClassifier` inspects each `ForLoopTree` / `WhileLoopTree` and returns `CanonicalFor`, `GenericWhile`, or `Other`. In `Translator.translate(StatementTree)`, after the existing `@BoundedBy` check, we call the classifier and emit `CAST.Statement.verbatim("bpf_for(...) { ... }")` on `CanonicalFor` or `CAST.Statement.verbatim("while (can_loop) { if (!(cond)) break; ... }")` on `GenericWhile`. A per-`Translator` flag `usedMayGoto` is set on either emit; `CompilerPlugin.combineCode` conditionally prepends `#include "bpf_experimental.h"` when set. The header is vendored into `bpf-compiler-plugin/src/main/resources/` and copied next to each generated `.c` file when the flag fires.

**Tech Stack:** Java 25, Maven, hello-ebpf compiler-plugin (`bpf-compiler-plugin`), CAST (`bpf-processor`), JUnit 5 for plugin unit tests, vng + real kernel for the smoke test.

**Spec:** `docs/superpowers/specs/2026-07-02-bpf-for-design.md`.

```c
/*
 * Trimmed subset of tools/testing/selftests/bpf/bpf_experimental.h from
 * the Linux kernel source tree. Contains only the may_goto machinery and
 * the bpf_for macro used by hello-ebpf's compiler plugin to lower
 * dynamic-bound loops. Retains the upstream SPDX identifier.
 *
 * See docs/superpowers/specs/2026-07-02-bpf-for-design.md for context.
 */

#ifndef __BPF_EXPERIMENTAL__
#define __BPF_EXPERIMENTAL__

#ifndef __arraycount
#define __arraycount(x) (sizeof(x) / sizeof((x)[0]))
#endif

/*
 * may_goto is a verifier-visible instruction that lets a loop fall
 * through to termination after a bounded (BPF_MAX_LOOPS ~ 8 million by
 * default) number of iterations, regardless of what the runtime
 * condition evaluates to.
 */
#define __may_goto_label_0 __PASTE(__may_goto_label_, __COUNTER__)

#define can_loop                                    \
	({ __label__ l_break, l_continue;           \
	   bool ret = true;                         \
	   asm volatile goto("may_goto %l[l_break]" \
	                     :::: l_break);         \
	   goto l_continue;                         \
	   l_break: ret = false;                    \
	   l_continue:;                             \
	   ret;                                     \
	})

#define may_goto                                    \
	({ __label__ l_break;                       \
	   asm volatile goto("may_goto %l[l_break]" \
	                     :::: l_break);         \
	   l_break:;                                \
	})

#define bpf_for(i, start, end) \
	for (i = (start); i < (end) && can_loop; i++)

#endif /* __BPF_EXPERIMENTAL__ */
```

```java
package me.bechberger.ebpf.bpf.compiler;

import com.sun.source.tree.*;

import java.util.function.Predicate;

/**
 * Classify a Java loop into one of the shapes the compiler plugin knows how to
 * lower using kernel `may_goto`-backed macros. See
 * docs/superpowers/specs/2026-07-02-bpf-for-design.md for the shape rules.
 */
public final class LoopShapeClassifier {

    private LoopShapeClassifier() {}

    /** Sealed shape hierarchy: pattern-match on the result. */
    public sealed interface LoopShape {
        /** for (T i = start; i {@literal <}(=) end; i++)  where at least one bound is dynamic. */
        record CanonicalFor(String var, ExpressionTree start, ExpressionTree end,
                            boolean inclusive) implements LoopShape {}
        /** while (cond)  where cond is dynamic. */
        record GenericWhile(ExpressionTree cond) implements LoopShape {}
        /** Anything else. Falls through to today's translation. */
        record Other() implements LoopShape {}
    }

    /**
     * Classify a for-loop.
     *
     * @param isConstant returns true iff the given expression is a Java compile-time
     *                   constant per JLS §15.28. Passed in so the classifier stays
     *                   independent of the Translator's javac context.
     */
    public static LoopShape classifyFor(ForLoopTree loop, Predicate<ExpressionTree> isConstant) {
        // Initializer: exactly one, of the form `int|long i = <start>`.
        var init = loop.getInitializer();
        if (init == null || init.size() != 1 || !(init.get(0) instanceof VariableTree v)) {
            return new LoopShape.Other();
        }
        String varName = v.getName().toString();
        String typeStr = v.getType().toString();
        if (!(typeStr.equals("int") || typeStr.equals("long"))) {
            return new LoopShape.Other();
        }
        ExpressionTree start = v.getInitializer();
        if (start == null) return new LoopShape.Other();

        // Condition: `i < <end>` or `i <= <end>` (same variable).
        var rawCond = loop.getCondition();
        if (rawCond instanceof ParenthesizedTree p) rawCond = p.getExpression();
        if (!(rawCond instanceof BinaryTree cmp)) return new LoopShape.Other();
        boolean inclusive;
        switch (cmp.getKind()) {
            case LESS_THAN -> inclusive = false;
            case LESS_THAN_EQUAL -> inclusive = true;
            default -> { return new LoopShape.Other(); }
        }
        var lhs = cmp.getLeftOperand();
        if (lhs instanceof ParenthesizedTree pp) lhs = pp.getExpression();
        if (!(lhs instanceof IdentifierTree id) || !id.getName().toString().equals(varName)) {
            return new LoopShape.Other();
        }
        ExpressionTree end = cmp.getRightOperand();

        // Update: exactly one, `i++` or `++i` on the same variable.
        var upd = loop.getUpdate();
        if (upd == null || upd.size() != 1) return new LoopShape.Other();
        if (!(upd.get(0) instanceof ExpressionStatementTree es)) return new LoopShape.Other();
        var updExpr = es.getExpression();
        if (!(updExpr instanceof UnaryTree u)) return new LoopShape.Other();
        if (u.getKind() != Tree.Kind.POSTFIX_INCREMENT && u.getKind() != Tree.Kind.PREFIX_INCREMENT) {
            return new LoopShape.Other();
        }
        var updOperand = u.getExpression();
        if (!(updOperand instanceof IdentifierTree updId)
                || !updId.getName().toString().equals(varName)) {
            return new LoopShape.Other();
        }

        // Body must not reassign the loop variable at the top level.
        if (bodyReassigns(loop.getStatement(), varName)) return new LoopShape.Other();

        // At least one of start/end must be non-constant.
        if (isConstant.test(start) && isConstant.test(end)) return new LoopShape.Other();

        return new LoopShape.CanonicalFor(varName, start, end, inclusive);
    }

    /**
     * Classify a while-loop.
     */
    public static LoopShape classifyWhile(WhileLoopTree loop, Predicate<ExpressionTree> isConstant) {
        var cond = loop.getCondition();
        if (cond instanceof ParenthesizedTree p) cond = p.getExpression();
        if (isConstant.test(cond)) return new LoopShape.Other();
        return new LoopShape.GenericWhile(cond);
    }

    /** Shallow check: does any top-level statement in the body reassign `var`? */
    private static boolean bodyReassigns(StatementTree body, String var) {
        if (body instanceof BlockTree block) {
            for (var s : block.getStatements()) {
                if (isAssignmentTo(s, var)) return true;
            }
            return false;
        }
        return isAssignmentTo(body, var);
    }

    private static boolean isAssignmentTo(StatementTree s, String var) {
        if (!(s instanceof ExpressionStatementTree es)) return false;
        var e = es.getExpression();
        if (e instanceof AssignmentTree a
                && a.getVariable() instanceof IdentifierTree id
                && id.getName().toString().equals(var)) return true;
        if (e instanceof CompoundAssignmentTree ca
                && ca.getVariable() instanceof IdentifierTree cid
                && cid.getName().toString().equals(var)) return true;
        return false;
    }
}
```

```java
/**
 * Demonstrates automatic bpf_for lowering. The @BPFFunction body contains
 * a canonical-shape for-loop with a dynamic bound (a GlobalVariable). Before
 * this feature landed, the verifier rejected such loops; now the compiler
 * plugin lowers them to bpf_for(...) so they load and run.

@BPF(license = "GPL")
public abstract class DynamicLoopSample extends BPFProgram implements SystemCallHooks {

    static final String EBPF_PROGRAM = """
            #include "vmlinux.h"
            #include <bpf/bpf_helpers.h>
            """;

    /** Runtime-varying upper bound, set from user space before attach. */
    public final GlobalVariable<Integer> n = new GlobalVariable<>(0);
```
